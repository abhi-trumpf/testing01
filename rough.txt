using System;

namespace CodingPractise
{
    public class ListNode
    {
        public int val;
        public ListNode? next;
        public ListNode(int val = 0, ListNode? next = null)
        {
            this.val = val;
            this.next = next;
        }
    }

    public class AddTwoLists
    {
        public static ListNode? AddTwoNumbers(ListNode? l1, ListNode? l2)
        {
            ListNode dummyHead = new ListNode(0);
            ListNode current = dummyHead;
            int carry = 0;

            while (l1 != null || l2 != null || carry != 0)
            {
                int val1 = (l1 != null) ? l1.val : 0;
                int val2 = (l2 != null) ? l2.val : 0;

                int sum = val1 + val2 + carry;
                carry = sum / 10;

                current.next = new ListNode(sum % 10);
                current = current.next;

                if (l1 != null) l1 = l1.next;
                if (l2 != null) l2 = l2.next;
            }

            return dummyHead.next;
        }

        public static void Run()
        {
            Console.WriteLine("--- Add Two Lists ---");
            ListNode l1 = new ListNode(2, new ListNode(4, new ListNode(3)));
            ListNode l2 = new ListNode(5, new ListNode(6, new ListNode(4)));

            Console.Write("List 1: ");
            PrintList(l1);
            Console.Write("List 2: ");
            PrintList(l2);

            ListNode? result = AddTwoNumbers(l1, l2);
            Console.Write("Result List: ");
            PrintList(result);
        }

        private static void PrintList(ListNode? head)
        {
            ListNode? temp = head;
            while (temp != null)
            {
                Console.Write(temp.val);
                if (temp.next != null) Console.Write(" -> ");
                temp = temp.next;
            }
            Console.WriteLine();
        }
    }
}

## Core Architecture & Memory Concepts
Value Types vs. Reference Types: Primitives (numbers, strings) sit in Stack memory and compare by raw value (5 === 5 is true). Objects and arrays sit in Heap memory and compare by physical RAM address location ([1,2,3] === [1,2,3] is false)

Implicit Type Coercion & Truthiness: The mechanism where [] == 0 is true because JavaScript forces the array into an empty string "" and then into the number 0. This creates the paradox where [] is structurally truthy on its own, but falsy in loose comparisons.

The Equality Operators (== vs ===): Strict equality (===) checks content and type honestly without alteration. Loose equality (==) forces background conversions and causes transitivity bugs (e.g., "" == 0 and [] == 0 are true, but 0 == null is false).

## Nullish Values & Math Safety
Nullish Primitives (undefined vs null): undefined is JavaScript’s default state for a variable that has no assigned value. null is a programmer's intentional assignment representing an explicitly empty state or a failed search (like a missing DOM element).

The Missing-Value Shortcut: How variable == null acts as a hardcoded language exception to catch both null and undefined at once, while safely ignoring valid falsy data like 0 or "".

NaN (Not a Number) & Math Safety: NaN belongs to the "number" data type as a mathematical error token. Passing missing data into math (Number(undefined)) outputs NaN, which aggressively infects any further mathematical equations.
